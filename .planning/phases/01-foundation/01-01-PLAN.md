<task type="auto">

  <files>None (remote operation)</files>
  <action>
    Use `gh repo create gesture-portfolio --public --source=. --remote=origin` to create the GitHub repo.
    This creates the repo on GitHub and sets up the remote origin.

  </action>

  </verify>

</task>

<task type="auto">
  <name>Task 2: Create project structure</name>
  <files>.gitignore, README.md, src/, src/js/, styles/, assets/, assets/images/</files>
  <action>
    Create standard static site structure:
    ```
    /
    ├── src/
    │   └── js/           # JavaScript modules
    ├── styles/           # CSS files
    ├── assets/
    │   └── images/       # Static assets
    ├── .gitignore        # Node modules, OS files, editor files
    └── README.md         # Project description
    ```

    .gitignore should include:
    - node_modules/
    - .DS_Store
    - *.log
    - .env
    - .vscode/ (optional, some prefer to keep)

    README.md should have:
    - Project title: Gesture-Controlled Portfolio
    - One-liner description
    - Tech stack list (HTML5, CSS3, Vanilla JS, Three.js, MediaPipe)
    - "Work in progress" note
  </action>
  <verify>
    - All directories exist: `ls -la src/js styles assets/images`
    - .gitignore exists and contains node_modules
    - README.md exists with content
  </verify>
  <done>Project structure created with .gitignore and README.md</done>
</task>

<task type="auto">
  <name>Task 3: Initial commit and push</name>
